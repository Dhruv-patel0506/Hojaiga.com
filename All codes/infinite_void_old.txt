        const voidSurface = document.getElementById('infinite-void');
        const statusText = document.getElementById('conn-status');
        
        // --- 1. INFINITE CAMERA / PANNING LOGIC ---
        let camera = { x: -50000, y: -50000, zoom: 1 }; // Start in the middle of the massive void
        let isDragging = false;
        let startDrag = { x: 0, y: 0 };

        function updateCamera() {
            voidSurface.style.transform = `translate(${camera.x}px, ${camera.y}px) scale(${camera.zoom})`;
        }

        document.getElementById('viewport').addEventListener('mousedown', (e) => {
            isDragging = true;
            startDrag = { x: e.clientX - camera.x, y: e.clientY - camera.y };
        });

        window.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            camera.x = e.clientX - startDrag.x;
            camera.y = e.clientY - startDrag.y;
            updateCamera();
        });

        window.addEventListener('mouseup', () => { isDragging = false; });
        
        // Center camera initially
        camera.x = (window.innerWidth / 2) - 1000; 
        camera.y = (window.innerHeight / 2) - 500;
        updateCamera();


        // --- 2. THE DYNAMIC RENDERING ENGINE ---
        const sleep = (ms) => new Promise(resolve => setTimeout(resolve, ms));

        // Settings for a very neat, human-like marker (Matches Image 3 and 8)
        const ROUGH_OPTIONS = {
            strokeWidth: 4,      // Thick marker
            roughness: 0.15,     // Very neat, almost no jitter
            bowing: 0.2,         // Very straight lines
            seed: 42,
            curveFitting: 0.95
        };

        async function renderElement(item) {
            // Absolute positioning within the massive void
            const baseX = 1000 + item.x; 
            const baseY = 500 + item.y;

            if (item.type === 'text' || item.type === 'math') {
                await drawTextOrMath(item, baseX, baseY);
            } else if (item.type === 'shape' || item.type === 'path') {
                await drawVector(item, baseX, baseY);
            }
        }

        async function drawTextOrMath(item, x, y) {
            const el = document.createElement('div');
            el.className = `board-element ${item.type === 'math' ? 'math-element' : 'text-element'}`;
            el.style.left = `${x}px`;
            el.style.top = `${y}px`;
            el.style.color = `var(--neon-${item.color || 'white'})`;
            el.style.opacity = 1;
            voidSurface.appendChild(el);

            if (item.type === 'math') {
                katex.render(item.content, el, { throwOnError: false, displayMode: true });
                el.style.opacity = 0;
                await sleep(100); 
                el.style.transition = "opacity 0.6s ease-in";
                el.style.opacity = 1;
                await sleep(500); 
            } else {
                for (let i = 0; i < item.content.length; i++) {
                    el.innerHTML += item.content.charAt(i);
                    await sleep(Math.random() * 20 + 20); // Fast, neat writing speed
                }
                await sleep(200);
            }
        }

        async function drawVector(item, x, y) {
            const svgNS = "http://www.w3.org/2000/svg";
            const svgEl = document.createElementNS(svgNS, "svg");
            svgEl.setAttribute("class", "board-element drawing-element");
            svgEl.style.left = `${x}px`;
            svgEl.style.top = `${y}px`;
            // Bounding box large enough for the drawing
            svgEl.setAttribute("width", "2000"); 
            svgEl.setAttribute("height", "2000");
            svgEl.style.opacity = 1;
            voidSurface.appendChild(svgEl);

            const rc = rough.svg(svgEl);
            const options = { ...ROUGH_OPTIONS, stroke: `var(--neon-${item.color || 'white'})` };
            
            let drawnNode = null;

            if (item.type === 'shape') {
                if (item.shape === 'line') drawnNode = rc.line(0, 0, item.w, item.h, options);
                if (item.shape === 'rect') drawnNode = rc.rectangle(0, 0, item.w, item.h, options);
                if (item.shape === 'circle') drawnNode = rc.circle(item.w/2, item.w/2, item.w, options);
                if (item.shape === 'ellipse') drawnNode = rc.ellipse(item.w/2, item.h/2, item.w, item.h, options);
            } else if (item.type === 'path') {
                // This allows CrewAI to send complex SVGs (like the Rocket or Tree)
                drawnNode = rc.path(item.svgPath, options);
            }

            if (drawnNode) {
                svgEl.appendChild(drawnNode);
                
                // Organic drawing animation
                const paths = drawnNode.querySelectorAll('path');
                let maxAnimTime = 0;

                paths.forEach((path, index) => {
                    const length = path.getTotalLength();
                    path.style.strokeDasharray = length;
                    path.style.strokeDashoffset = length;
                    path.style.stroke = options.stroke;
                    
                    const duration = Math.max(0.4, length / 400); 
                    const delay = index * 0.15; 
                    maxAnimTime = Math.max(maxAnimTime, (duration + delay) * 1000);

                    path.style.animation = `drawStroke ${duration}s ease-in-out ${delay}s forwards`;
                });

                await sleep(maxAnimTime + 100); 
            }
        }


        // --- 3. CREW AI INTERFACE (HOW YOUR BACKEND CONNECTS) ---

        let aiSocket = null;

        /* 
         * Connect to Python AI backend via WebSocket
         */
        function connectToAI() {
            const wsUrl = 'ws://localhost:8765';
            aiSocket = new WebSocket(wsUrl);
            
            aiSocket.onopen = () => {
                statusText.innerText = "Connected to AI Server";
                statusText.style.color = "#39ff14";
                console.log("Connected to AI server");
            };
            
            aiSocket.onmessage = async (event) => {
                const aiCommand = JSON.parse(event.data);
                
                // Handle different command types
                if (aiCommand.type === 'connection_ok') {
                    console.log("Server connection confirmed:", aiCommand.message);
                }
                else if (aiCommand.type === 'lesson_complete') {
                    statusText.innerText = "Lesson complete! Drag around to explore.";
                    statusText.style.color = "#fffb00";
                }
                // If AI commands the camera to move to a new area of the void
                else if (aiCommand.action === 'move_camera') {
                    camera.x -= aiCommand.x;
                    camera.y -= aiCommand.y;
                    updateCamera();
                    await sleep(1000); // Wait for pan
                } 
                // If AI commands to draw
                else {
                    await renderElement(aiCommand);
                }
            };
            
            aiSocket.onerror = (error) => {
                statusText.innerText = "Connection failed - check if server is running";
                statusText.style.color = "#ff0000";
                console.error("WebSocket error:", error);
            };
            
            aiSocket.onclose = () => {
                statusText.innerText = "Disconnected from AI Server";
                statusText.style.color = "#ff0000";
            };
        }

        function requestLesson() {
            const topicInput = document.getElementById('topic-input');
            const topic = topicInput.value.trim();
            
            if (!topic) {
                alert("Please enter a topic");
                return;
            }
            
            if (!aiSocket || aiSocket.readyState !== WebSocket.OPEN) {
                alert("Not connected to AI server. Please wait for connection...");
                return;
            }
            
            statusText.innerText = "Requesting lesson...";
            statusText.style.color = "#00ffff";
            
            // Send lesson request to AI
            aiSocket.send(JSON.stringify({
                type: "start_lesson",
                topic: topic
            }));
        }

        // Auto-connect on page load
        window.addEventListener('load', () => {
            setTimeout(connectToAI, 1000);
        });

### How this works dynamically with CrewAI & Existing Packages:

1.  **The New `path` Type:** Look at the "Velocity vector" in the mock payload. Instead of basic lines, the AI sent an `svgPath: "M 0 0 L 100 -50..."`. This is incredibly powerful.
    *   If you have a **Python package that plots graphs** (like Matplotlib or NetworkX), you can instruct CrewAI to generate an SVG string from that graph, put it in the JSON payload, and this HTML engine will magically draw it in neon on the lightboard.
2.  **The Infinite Space:** You can click and drag this entire canvas with your mouse. Because the internal `div` is massive (`100,000px`), the AI can write a whole physics lesson, then send a `move_camera` command via WebSocket, slide the screen over to a blank area, and start drawing the next topic. 
3.  **Colors:** I mapped the exact neon colors from your images (`cyan`, `magenta`, `green`, `yellow`, `white`, `orange`). Your CrewAI prompt simply needs to tell the JSON which color to use.

This file is now a completely un-opinionated, infinite renderer. It does not dictate *what* is drawn; it relies entirely on the intelligence of your local 26B/34B model to send the coordinates and text.
