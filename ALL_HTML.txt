    # ─────────────────────────────────────────────────────────────────────────────
    # 3-TIER AUTONOMOUS ERROR RECOVERY ENGINE
    # ─────────────────────────────────────────────────────────────────────────────
    async def _execute_autonomous_recovery(
        self,
        action: str,
        parameters: dict,
        session_id: str,
        agent_id: str,
        task_id: str,
        error_detail: str,
        is_browser_cmd: bool,
        task_objective: str = "",
    ) -> str:
        """
        Orchestrates 3 escalating autonomous recovery tiers on failure:
        Tier 1: Visual Grounding Retry (locate element visually via VLM & click)
        Tier 2: Cognitive Blocker Diagnosis (detect & clear popups/dialogs/banners)
        Tier 3: Adaptive Replanner (consult local Brain LLM to propose alternative actions)
        """
        log.info(f"Agent {agent_id}: [AUTONOMOUS RECOVERY] Initiated for failed action '{action}'...")

        # ── TIER 1: VISUAL GROUNDING RETRY ─────────────────────────────────────
        if is_browser_cmd and action in ("CLICK", "TYPE", "SCROLL", "INTERACT"):
            log.info(f"Agent {agent_id}: [TIER 1 RECOVERY] Attempting visual grounding retry via VLM...")
            res1 = await self._visual_grounding_retry(action, parameters, session_id, agent_id, task_id, error_detail)
            if res1 == "RECOVERED":
                return "RECOVERED"
            log.warning(f"Agent {agent_id}: [TIER 1 RECOVERY] Visual grounding failed: {res1}")

        # ── TIER 2: COGNITIVE BLOCKER DIAGNOSIS & CLEARANCE ────────────────────
        log.info(f"Agent {agent_id}: [TIER 2 RECOVERY] Analyzing screen for modals/popups/blockers...")
        cleared = await self._diagnose_and_clear_blocker(action, parameters, session_id, agent_id, task_id, error_detail, is_browser_cmd)
        if cleared:
            log.info(f"Agent {agent_id}: [TIER 2 RECOVERY] Blocker cleared! Re-attempting action '{action}'...")
            if is_browser_cmd and action in ("CLICK", "TYPE", "SCROLL", "INTERACT"):
                res_retry = await self._visual_grounding_retry(action, parameters, session_id, agent_id, task_id, error_detail)
                if res_retry == "RECOVERED":
                    return "RECOVERED_AFTER_BLOCKER_CLEARANCE"
            return "RECOVERED_AFTER_BLOCKER_CLEARANCE"

        # ── TIER 3: ADAPTIVE COGNITIVE REPLANNER ───────────────────────────────
        log.info(f"Agent {agent_id}: [TIER 3 RECOVERY] Invoking cognitive replanner for alternative sub-actions...")
        replanned = await self._replan_failed_action(action, parameters, session_id, agent_id, task_id, error_detail, task_objective)
        if replanned:
            return "RECOVERED_VIA_REPLANNING"

        # Record failure into AppKnowledgeManager for future cross-run awareness
        try:
            from core_organs.organ_knowledge_manager import AppKnowledgeManager
            km = AppKnowledgeManager.get_instance()
            target_name = parameters.get("target") or parameters.get("app_name") or action
            km.record_action_failure(target_name, action, str(parameters)[:40], error_detail)
        except Exception:
            pass

        return "FAILED_ALL_TIERS"

    async def _diagnose_and_clear_blocker(
        self,
        action: str,
        parameters: dict,
        session_id: str,
        agent_id: str,
        task_id: str,
        error_detail: str,
        is_browser_cmd: bool,
    ) -> bool:
        """Tier 2: Asks VLM if a popup/modal/cookie banner is blocking and dismisses it."""
        try:
            from core_organs.lead_extractor import send_cdp_screenshot, send_cdp_click_coordinates, send_cdp_command
            from core_organs.organ_31_visual_perception import VisualGroundingEngine
            import requests as _req

            screenshot_b64 = None
            if is_browser_cmd:
                screenshot_b64 = send_cdp_screenshot(format="jpeg", quality=80, timeout=15.0)
            if not screenshot_b64:
                try:
                    from PIL import ImageGrab
                    import io, base64
                    img = ImageGrab.grab()
                    buf = io.BytesIO()
                    img.save(buf, format="JPEG", quality=80)
                    screenshot_b64 = base64.b64encode(buf.getvalue()).decode("utf-8")
                except Exception:
                    return False

            if not screenshot_b64:
                return False

            grounding = VisualGroundingEngine()
            blocker_prompt = (
                f"The automation agent tried action '{action}', but failed with error: '{error_detail}'.\n"
                "Look closely at this screen image:\n"
                "1. Is there an obstacle, modal dialog, popup, cookie banner, or overlay blocking the page?\n"
                "2. If yes, what button will dismiss it (e.g. 'Accept All', 'Close', 'Dismiss', 'Agree', 'Cancel', 'X')?\n\n"
                "Respond ONLY in JSON format:\n"
                "{\n"
                '  "is_blocked": true,\n'
                '  "blocker_name": "name of blocker",\n'
                '  "dismiss_button_text": "label of button to click",\n'
                '  "suggest_escape_key": false\n'
                "}"
            )
            payload = {
                "model": grounding.model_name,
                "messages": [{"role": "user", "content": blocker_prompt, "images": [screenshot_b64]}],
                "stream": False,
                "options": {"temperature": 0.1}
            }
            resp = _req.post(f"{grounding.ollama_url.rstrip('/')}/api/chat", json=payload, timeout=25)
            if resp.status_code != 200:
                return False

            content = resp.json().get("message", {}).get("content", "")
            match_json = re.search(r"\{[\s\S]*\}", content)
            if not match_json:
                return False
            data = json.loads(match_json.group(0))

            if not data.get("is_blocked"):
                return False

            blocker_name = data.get("blocker_name", "modal overlay")
            dismiss_text = data.get("dismiss_button_text", "")
            use_escape = data.get("suggest_escape_key", False)

            log.info(f"Agent {agent_id}: [BLOCKER DETECTED] Found '{blocker_name}'. Clearing...")

            if use_escape:
                if is_browser_cmd:
                    send_cdp_command(0x0151, {"type": "keyDown", "key": "Escape", "windowsVirtualKeyCode": 27}, timeout=5.0)
                    send_cdp_command(0x0151, {"type": "keyUp", "key": "Escape", "windowsVirtualKeyCode": 27}, timeout=5.0)
                await asyncio.sleep(1.0)
                return True

            if dismiss_text:
                match = grounding.locate_element(dismiss_text, image_base64=screenshot_b64)
                if match.found:
                    if is_browser_cmd:
                        send_cdp_click_coordinates(match.pixel_x, match.pixel_y, timeout=8.0)
                    log.info(f"Agent {agent_id}: [BLOCKER CLEARED] Clicked '{dismiss_text}' at ({match.pixel_x}, {match.pixel_y})")
                    await asyncio.sleep(1.5)
                    return True
        except Exception as e:
            log.warning(f"Agent {agent_id}: Blocker diagnosis failed: {e}")
        return False

    async def _replan_failed_action(
        self,
        action: str,
        parameters: dict,
        session_id: str,
        agent_id: str,
        task_id: str,
        error_detail: str,
        task_objective: str,
    ) -> bool:
        """Tier 3: Dynamic Adaptive Replanner via local LLM."""
        try:
            from core_organs.organ_37_llm_gateway import LLMGateway
            gw = LLMGateway()
            replan_prompt = (
                f"You are the autonomous replanning engine for Hojaiga OS.\n"
                f"Task Objective: '{task_objective}'\n"
                f"Failed Step: '{action}' with parameters {json.dumps(parameters)}\n"
                f"Failure Error: '{error_detail}'\n\n"
                "Formulate an alternative workaround plan (1 or 2 actions max) to accomplish the goal.\n"
                "Allowed actions: NAVIGATE, CLICK, TYPE, PRESS_KEY, SCROLL, VISUAL_CLICK, LAUNCH_APP.\n"
                "Respond ONLY in JSON:\n"
                "{\n"
                '  "strategy": "concise explanation of workaround",\n'
                '  "alternate_actions": [\n'
                '    {"action": "...", "parameters": {...}}\n'
                '  ]\n'
                "}"
            )
            raw = gw.execute_small_inference(
                system="You are an autonomous fallback replanner. Respond strictly in valid JSON.",
                prompt=replan_prompt,
                timeout=25.0
            )
            match_json = re.search(r"\{[\s\S]*\}", raw)
            if not match_json:
                return False
            plan_data = json.loads(match_json.group(0))
            alt_actions = plan_data.get("alternate_actions", [])
            strategy = plan_data.get("strategy", "alternative path")

            if not alt_actions:
                return False

            log.info(f"Agent {agent_id}: [TIER 3 REPLANNING] Executing strategy: {strategy}")
            for alt in alt_actions:
                alt_act = alt.get("action", "").upper()
                alt_params = alt.get("parameters", {})
                if alt_act == "NAVIGATE" and alt_params.get("url"):
                    from core_organs.lead_extractor import send_cdp_navigate
                    send_cdp_navigate(alt_params["url"], timeout=15.0)
                    log.info(f"Agent {agent_id}: Replan navigated to: {alt_params['url']}")
                elif alt_act in ("VISUAL_CLICK", "CLICK") and alt_params.get("description"):
                    from core_organs.organ_31_visual_perception import VisualGroundingEngine
                    from core_organs.lead_extractor import send_cdp_screenshot, send_cdp_click_coordinates
                    grounding = VisualGroundingEngine()
                    b64 = send_cdp_screenshot(format="jpeg", quality=80, timeout=10.0)
                    if b64:
                        m = grounding.locate_element(alt_params["description"], image_base64=b64)
                        if m.found:
                            send_cdp_click_coordinates(m.pixel_x, m.pixel_y, timeout=8.0)
                            log.info(f"Agent {agent_id}: Replan visual click at ({m.pixel_x}, {m.pixel_y})")
                elif alt_act == "PRESS_KEY" and alt_params.get("key"):
                    from core_organs.lead_extractor import send_cdp_command
                    k = alt_params["key"]
                    if k == "Enter":
                        send_cdp_command(0x0151, {"type": "keyDown", "key": "Enter", "windowsVirtualKeyCode": 13}, timeout=5.0)
                        send_cdp_command(0x0151, {"type": "keyUp", "key": "Enter", "windowsVirtualKeyCode": 13}, timeout=5.0)
                await asyncio.sleep(1.0)
            return True
        except Exception as e:
            log.warning(f"Agent {agent_id}: Adaptive replanning failed: {e}")
        return False
    async def _visual_grounding_retry(
        self,
        action: str,
        parameters: dict,
        session_id: str,
        agent_id: str,
        task_id: str,
        error_detail: str,
    ) -> str:
        """
        When a DOM-based browser action fails, attempt recovery via visual grounding:
        1. Capture CDP screenshot
        2. Ask qwen2.5vl:3b to locate the target element by natural language
        3. Click via CDP pixel coordinates
        4. Verify state changed with a post-action screenshot
        Returns "RECOVERED" or failure reason string.
        """
        try:
            from core_organs.lead_extractor import send_cdp_screenshot, send_cdp_click_coordinates, send_cdp_js
            from core_organs.organ_31_visual_perception import VisualGroundingEngine
        except ImportError as e:
            return f"visual imports unavailable: {e}"

        grounding = VisualGroundingEngine()

        # 1. Capture screenshot from active Chrome tab
        screenshot_b64 = send_cdp_screenshot(format="jpeg", quality=80, timeout=15.0)
        if not screenshot_b64:
            return "CDP screenshot capture failed"

        # 2. Build a natural-language target description from the failed action
        target_desc = self._build_visual_target(action, parameters, error_detail)
        log.info(f"Agent {agent_id}: Visual grounding retry — looking for: '{target_desc}'")

        # 3. Locate element visually via qwen2.5vl:3b
        match = grounding.locate_element(target_desc, image_base64=screenshot_b64)
        if not match.found:
            return f"VLM could not locate element: {match.reason}"

        log.info(f"Agent {agent_id}: Visual match at ({match.pixel_x}, {match.pixel_y}), box={match.box_2d}")

        # 4. Capture BEFORE state for verification
        before_b64 = screenshot_b64

        # 5. Click via CDP pixel coordinates
        if action in ("CLICK", "INTERACT"):
            click_ok = send_cdp_click_coordinates(match.pixel_x, match.pixel_y, timeout=10.0)
            if not click_ok:
                return f"CDP click at ({match.pixel_x}, {match.pixel_y}) failed"
            log.info(f"Agent {agent_id}: Visual click dispatched to ({match.pixel_x}, {match.pixel_y})")

        elif action == "TYPE":
            # Click the input field first, then type
            click_ok = send_cdp_click_coordinates(match.pixel_x, match.pixel_y, timeout=10.0)
            if not click_ok:
                return f"CDP click for TYPE at ({match.pixel_x}, {match.pixel_y}) failed"
            text_to_type = parameters.get("text", "")
            if text_to_type:
                import time as _time
                _time.sleep(0.3)
                # Use CDP Input.insertText for reliable text entry
                from core_organs.lead_extractor import send_cdp_command
                send_cdp_command(0x0150, {"text": text_to_type}, timeout=10.0)
                log.info(f"Agent {agent_id}: Visual type '{text_to_type[:40]}' at ({match.pixel_x}, {match.pixel_y})")

        elif action == "SCROLL":
            # Scroll is usually not element-specific, but we can try JS scroll
            send_cdp_js("window.scrollBy(0, 600);", timeout=8.0)

        # 6. Post-action screenshot + visual verification
        import time as _time
        _time.sleep(1.5)
        after_b64 = send_cdp_screenshot(format="jpeg", quality=80, timeout=15.0)
        if after_b64:
            verification = grounding.verify_state_change(
                action_description=f"Action '{action}' on '{target_desc}'",
                before_b64=before_b64,
                after_b64=after_b64,
            )
            log.info(f"Agent {agent_id}: Visual verification: {verification}")

        return "RECOVERED"

    def _build_visual_target(self, action: str, parameters: dict, error_detail: str) -> str:
        """Build a natural-language description of what the vision model should find."""
        # If there's a 'text' parameter (e.g. a button label), use it
        text = parameters.get("text", "")
        target = parameters.get("target", "")
        url = parameters.get("url", "")

        parts = []
        if text:
            parts.append(f'element with text "{text[:60]}"')
        if target:
            parts.append(f'element matching "{target[:60]}"')

        if action == "CLICK":
            if parts:
                return f"clickable button or link: {', '.join(parts)}"
            return "the most prominent clickable button or link on screen"
        elif action == "TYPE":
            if parts:
                return f"text input field: {', '.join(parts)}"
            return "an active text input field or search box"
        elif action == "SCROLL":
            return "scrollable content area or feed"
        elif action == "NAVIGATE":
            return f"address bar or URL field for navigation to {url}"
        else:
            if parts:
                return f"UI element: {', '.join(parts)}"
            return f"relevant UI element for action '{action}'"

    # ─────────────────────────────────────────────────────────────────────────────
    # Native Organ 10 Capabilities: Video Intelligence & Multilingual Synthesis
    # ─────────────────────────────────────────────────────────────────────────────
    def process_video_and_synthesize_notes(
        self,
        video_url_or_id: str,
        topic_context: str = "",
        output_format: str = "all",  # "md", "docx", "pptx", or "all"
        output_dir: str = "./data/notes"
    ) -> Dict[str, Any]:
        """Native Organ 10 function for Video-to-Notes synthesis:
        - Extracts transcript in ANY language (auto-translates to English).
        - Decodes spoken regional/mathematical idioms ('1 by 2' -> 1/2, '1 into 2' -> 1*2).
        - Pedagogical Signal-to-Noise Filtering: Discards filler/rambling; extracts real teaching content,
          theorems, derivations, architecture diagrams, and core takeaways.
        - Automatically exports to Markdown, Word (.docx), or PowerPoint (.pptx).
        """
        os.makedirs(output_dir, exist_ok=True)
        raw_items = VideoIntelligenceHelper.fetch_transcript(video_url_or_id)
        if not raw_items:
            return {"status": "error", "message": f"No transcript available for {video_url_or_id}"}

        chunks = VideoIntelligenceHelper.process_and_chunk(raw_items)
        title = topic_context or f"Video Analytical Synthesis ({video_url_or_id})"
        notes_md = VideoIntelligenceHelper.generate_teaching_notes(title, chunks)

        res = {"status": "success", "title": title, "markdown_length": len(notes_md), "chunks": len(chunks)}
        
        # Save Markdown
        safe_title = re.sub(r'[^a-zA-Z0-9_-]', '_', title)[:40]
        md_path = os.path.join(output_dir, f"{safe_title}.md")
        with open(md_path, "w", encoding="utf-8") as f:
            f.write(notes_md)
        res["markdown_file"] = md_path

        # Export DOCX
        if output_format in ["docx", "all"]:
            docx_path = os.path.join(output_dir, f"{safe_title}.docx")
            VideoIntelligenceHelper.export_to_docx(notes_md, docx_path)
            res["docx_file"] = docx_path

    def spreadsheet_operation(self, file_path: str, action: str, sheet_name: str = "Sheet1",
                              data: Optional[List[Dict[str, Any]]] = None, formula: Optional[str] = None) -> Dict[str, Any]:
        """Reads/writes Excel spreadsheets, calculates formulas, and appends structured rows."""
        import openpyxl
        p = Path(file_path)
        p.parent.mkdir(parents=True, exist_ok=True)
        
        if action == "create_or_append":
            if p.exists():
                wb = openpyxl.load_workbook(file_path)
                ws = wb[sheet_name] if sheet_name in wb.sheetnames else wb.create_sheet(sheet_name)
            else:
                wb = openpyxl.Workbook()
                ws = wb.active if wb.active is not None else wb.create_sheet(sheet_name)
                ws.title = sheet_name
                if data and len(data) > 0:
                    ws.append(list(data[0].keys())) # Header row
                    
            if data and ws is not None:
                for row_dict in data:
                    ws.append(list(row_dict.values()))
            wb.save(file_path)
            return {"status": "success", "file": file_path, "rows_written": len(data or [])}

        elif action == "read":
            if not p.exists():
                return {"status": "error", "message": f"File {file_path} does not exist"}
            wb = openpyxl.load_workbook(file_path, data_only=True)
            ws = wb[sheet_name] if sheet_name in wb.sheetnames else (wb.active or (wb.worksheets[0] if wb.worksheets else None))
            if ws is None:
                return {"status": "error", "message": f"Workbook in {file_path} contains no readable sheets"}
            rows = list(ws.iter_rows(values_only=True))
            headers = [str(h) for h in rows[0]] if rows else []
            records = [dict(zip(headers, r)) for r in rows[1:]] if len(rows) > 1 else []
            return {"status": "success", "total_rows": len(records), "data": records}

        return {"status": "unsupported_action"}

    # 3. File Intelligence & Organizer (Auto-Sort, Rename, Organize)
    def organize_directory(self, target_directory: str, mode: str = "by_extension") -> Dict[str, Any]:
        """Automatically scans, categorizes, and organizes cluttered directories."""
        import shutil
        p = Path(target_directory)
        if not p.exists():
            return {"status": "error", "message": f"Directory {target_directory} not found"}
            
        moved_count = 0
        for item in list(p.iterdir()):
            if item.is_file():
                ext = item.suffix.lower().lstrip(".") or "other"
                cat_folder = p / ext.upper()
                cat_folder.mkdir(exist_ok=True)
                dest = cat_folder / item.name
                if not dest.exists():
                    shutil.move(str(item), str(dest))
                    moved_count += 1
                    
        return {"status": "success", "directory": target_directory, "files_organized": moved_count}

    # 5. Universal API Caller (REST / JSON / Auth)
    def execute_api_call(self, url: str, method: str = "GET", headers: Optional[Dict[str, str]] = None,
                         json_payload: Optional[Dict[str, Any]] = None, timeout: float = 15.0) -> Dict[str, Any]:
        """Calls any external or internal HTTP/REST endpoint asynchronously or synchronously."""
        import urllib.request
        data_bytes = json.dumps(json_payload).encode() if json_payload else None
        h = {"Content-Type": "application/json", "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)", **(headers or {})}
        req = urllib.request.Request(url, data=data_bytes, headers=h, method=method.upper())
        try:
            with urllib.request.urlopen(req, timeout=timeout) as resp:
                res_data = resp.read().decode("utf-8")
                try:
                    parsed = json.loads(res_data)
                except Exception:
                    parsed = res_data
                return {"status": "success", "code": resp.status, "data": parsed}
        except Exception as e:
            return {"status": "error", "error": str(e)}

    # 6. PDF Intelligence (Extraction, Summarization, Form Parsing)
    def extract_pdf_intelligence(self, pdf_path: str, max_pages: int = 100) -> Dict[str, Any]:
        """Reads, parses, and extracts structured text and tables from PDF documents."""
        from pypdf import PdfReader
        p = Path(pdf_path)
        if not p.exists():
            return {"status": "error", "message": f"PDF {pdf_path} does not exist"}
        reader = PdfReader(str(p))
        total_pages = len(reader.pages)
        pages_to_read = min(total_pages, max_pages)
        full_text = []
        for idx in range(pages_to_read):
            page_text = reader.pages[idx].extract_text() or ""
            full_text.append(f"[Page {idx+1}]\n{page_text}")
            
        combined = "\n\n".join(full_text)
        return {
            "status": "success",
            "file": pdf_path,
            "total_pages": total_pages,
            "pages_parsed": pages_to_read,
            "character_count": len(combined),
            "sample_content": combined[:500]
        }

    SPOKEN_MATH_RULES = [
        (r'\b(\d+(?:\.\d+)?)\s+by\s+(\d+(?:\.\d+)?)\b', r'\1/\2'),
        (r'\b(\d+(?:\.\d+)?)\s+into\s+(\d+(?:\.\d+)?)\b', r'\1 * \2'),
        (r'\b([a-zA-Z])\s+by\s+([a-zA-Z])\b', r'\1/\2'),
        (r'\b([a-zA-Z])\s+into\s+([a-zA-Z])\b', r'\1 * \2'),
        (r'\bdivided\s+by\b', r'/'),
        (r'\bmultiplied\s+by\b', r'*'),
        (r'\bplus\s+or\s+minus\b', r'±'),
        (r'\bsquare\s+root\s+of\s+(\d+|[a-zA-Z])\b', r'√\1'),
        (r'\b([a-zA-Z0-9]+)\s+squared?\b', r'\1²'),
        (r'\b([a-zA-Z0-9]+)\s+cubed?\b', r'\1³'),
        (r'\b(\d+)\s+(?:lakh|lac|lakhs)\b', r'\1,00,00,000'),
        (r'\b(\d+)\s+(?:crore|crores)\b', r'\1,00,00,000'),
    ]

    @classmethod
    def normalize_idioms(cls, text: str) -> str:
        res = text
        for pattern, replacement in cls.SPOKEN_MATH_RULES:
            res = re.sub(pattern, replacement, res, flags=re.IGNORECASE)
        return res

    @classmethod
    def fetch_transcript(cls, video_url_or_id: str) -> List[Dict[str, Any]]:
        import youtube_transcript_api
        from youtube_transcript_api import YouTubeTranscriptApi
        vid = video_url_or_id
        if "v=" in video_url_or_id:
            vid = video_url_or_id.split("v=")[1].split("&")[0]
        elif "youtu.be/" in video_url_or_id:
            vid = video_url_or_id.split("youtu.be/")[1].split("?")[0]

        try:
            api = YouTubeTranscriptApi() if isinstance(YouTubeTranscriptApi, type) else YouTubeTranscriptApi
            # Try fetching list first
            if hasattr(api, "list"):
                tl = api.list(vid)
            elif hasattr(YouTubeTranscriptApi, "list_transcripts"):
                tl = YouTubeTranscriptApi.list_transcripts(vid)
            else:
                tl = None

            if tl:
                try:
                    t = tl.find_transcript(['en', 'en-US', 'hi', 'es', 'fr', 'de'])
                except Exception:
                    t = next(iter(tl))
                if t.language_code != 'en' and t.is_translatable:
                    t = t.translate('en')
                fetched = t.fetch()
                return [{"text": getattr(item, "text", str(item)), "start": getattr(item, "start", 0.0), "duration": getattr(item, "duration", 0.0)} for item in fetched]
            
            # Direct fetch fallback
            if hasattr(api, "fetch"):
                fetched = api.fetch(vid)
                return [{"text": getattr(item, "text", str(item)), "start": getattr(item, "start", 0.0), "duration": getattr(item, "duration", 0.0)} for item in fetched]
            elif hasattr(YouTubeTranscriptApi, "get_transcript"):
                return YouTubeTranscriptApi.get_transcript(vid)
            return []
        except Exception as e:
            log.warning(f"Transcript extraction failed: {e}")
            return []

    @classmethod
    def process_and_chunk(cls, raw_transcript: List[Dict[str, Any]], chunk_duration_sec: float = 600.0) -> List[Dict[str, Any]]:
        chunks = []
        cur_start: float = 0.0
        cur_end: float = 0.0
        cur_text: List[str] = []

        for item in raw_transcript:
            start = float(item.get("start", 0.0))
            duration = float(item.get("duration", 0.0))
            text = cls.normalize_idioms(str(item.get("text", "")))
            if not cur_text:
                cur_start = start
            cur_text.append(text)
            cur_end = start + duration
            if (cur_end - cur_start) >= chunk_duration_sec:
                chunks.append({
                    "time_range": f"{int(cur_start // 60)}m-{int(cur_end // 60)}m",
                    "content": " ".join(cur_text)
                })
                cur_start = cur_end
                cur_text = []
        if cur_text:
            chunks.append({
                "time_range": f"{int(cur_start // 60)}m-{int(cur_end // 60)}m",
                "content": " ".join(cur_text)
            })
        return chunks
